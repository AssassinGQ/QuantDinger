```python
def test_e2e_get_kline_returns_correct_format():
    """E2E: DataSourceFactory → IBKRDataSource → mocked ib_insync returns correct kline format."""
    import ib_insync

    # Create mock bar objects (ib_insync returns bar objects with .date, .open, etc.)
    mock_bars = []
    base_time = 1700000000
    for i in range(5):
        bar = MagicMock()
        bar.date = datetime.fromtimestamp(base_time + i * 60, tz=timezone.utc)
        bar.open = 150.0 + i
        bar.high = 151.0 + i
        bar.low = 149.0 + i
        bar.close = 150.5 + i
        bar.volume = 1000000 + i * 1000
        mock_bars.append(bar)

    # Create mock IB instance
    mock_ib_instance = MagicMock()
    mock_ib_instance.isConnected.return_value = True
    mock_ib_instance.reqHistoricalBarsAsync = AsyncMock(return_value=mock_bars)
    mock_ib_instance.qualifyContractAsync = AsyncMock(return_value=True)

    with patch('ib_insync.IB', return_value=mock_ib_instance):
        # Patch at the module where IBKRClient uses it
        with patch('app.services.live_trading.ibkr_trading.client.ib_insync.IB', return_value=mock_ib_instance):
            from app.data_sources import DataSourceFactory
            # Clear any cached source
            DataSourceFactory._sources.pop('ibkr-live', None)

            source = DataSourceFactory.get_source('USStock', exchange_id='ibkr-live')
            result = source.get_kline(symbol='AAPL', timeframe='1m', limit=5)

    # Verify format
    assert isinstance(result, list)
    assert len(result) == 5
    for bar in result:
        assert 'time' in bar
        assert 'open' in bar
        assert 'high' in bar
        assert 'low' in bar
        assert 'close' in bar
        assert 'volume' in bar
        assert isinstance(bar['time'], int)
        assert isinstance(bar['open'], float)
```
